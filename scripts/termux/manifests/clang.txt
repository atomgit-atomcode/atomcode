# clang + lld provider: what goes into the APK (executables) and what becomes the
# `clang-sysroot` data package (read-only headers and the resource directory).
#
# `bin/clang-21` is the driver; Termux links it against libLLVM.so and
# libclang-cpp.so, so the binary itself is small and its runtime closure is pulled
# in automatically by split.sh.
#
# `lib/clang/21` is clang's resource directory (builtin headers, per-target
# builtins). It is only read, so it ships as data — but the wrapper has to point
# clang at it with `-resource-dir`, because clang would otherwise look next to its
# own binary, which lives in nativeLibraryDir.
# `clang` itself is NOT in this list: it is the launcher compiled from
# scripts/termux/launcher/clang.c, which injects `-resource-dir` and `--sysroot`
# for the data package and then execs this driver. The driver needs those flags
# because it would otherwise look for its resource directory next to itself — in
# the APK's nativeLibraryDir, where the data package is not.
exec bin/clang-21 clang-real
exec bin/lld lld
# The archiver is part of the toolchain, not an extra: cmake looks `ar` up by name
# and refuses to configure with `CMAKE_AR-NOTFOUND` when it is absent (measured on
# device). `ranlib` likewise, for static libraries.
exec bin/llvm-ar ar
exec bin/llvm-ranlib ranlib
# The language server. Without it the editor-side features (diagnostics, hover,
# definitions) have nothing to talk to on a device where no compiler package can
# be installed — and it has to ride in the APK, because only `nativeLibraryDir`
# may be executed.
#
# The stem is `clangd-real`, not `clangd`: the launcher compiled from
# scripts/termux/launcher/clangd.c owns `libclangd.so` (that is the name on PATH),
# and the script copies jniLibs before it compiles launchers. Naming this one
# `clangd` made the launcher overwrite the 18 MB server — the split still reported
# five executables, so the loss was invisible until the launcher could not find
# `libclangd-real.so` on device.
exec bin/clangd clangd-real

data include
data lib/clang
